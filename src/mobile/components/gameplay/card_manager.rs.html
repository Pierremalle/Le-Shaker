1use crate::config::questions::QuestionDeck;
2use crate::config::state::next_player;
3use dioxus::prelude::*;
4
5/// Component used to display the different cards during the game
6/// Use the player and question list to display it and change infinitly
7#[component]
8pub fn CardManager() -> Element {
9    let mut deck = use_signal(|| QuestionDeck::new());
10
11    let mut current_card = use_signal(|| None::<(String, String, String)>);
12    let mut current_player = use_signal(|| None::<String>);
13
14    use_effect(move || {
15        let card = deck.write().next_card();
16
17        current_card.set(Some((
18            card.category.to_string(),
19            card.question.to_string(),
20            card.alternative.to_string(),
21        )));
22
23        let player = next_player().unwrap_or_else(|| "Aucun joueur".to_string());
24
25        current_player.set(Some(player));
26    });
27
28    // anonymous function used to load a new card and change player
29    let load_next = move |_| {
30        let card = deck.write().next_card();
31
32        current_card.set(Some((
33            card.category.to_string(),
34            card.question.to_string(),
35            card.alternative.to_string(),
36        )));
37
38        current_player.set(Some(
39            next_player().unwrap_or_else(|| "Aucun joueur".to_string()),
40        ));
41    };
42
43    rsx! {
44        div { class: "flex flex-col items-center gap-6 w-full",
45
46            // if a card exit, display it with targeted player, else a loading text
47            if let Some((category, question, alternative)) = current_card() {
48                div { class: "w-full h-64 rounded-3xl bg-gradient-to-br from-purple-700 to-pink-600 p-6 shadow-[0_0_20px_rgba(168,85,247,0.8)] flex flex-col justify-between",
49
50                    if let Some(player) = current_player() {
51                        h2 { class: "text-xl font-bold text-purple-200", "🎯 {player}" }
52                    }
53
54                    div {
55                        p { class: "text-sm opacity-70", "Catégorie : {category}" }
56                        p { class: "text-lg mt-2", "{question}" }
57                        if alternative != "" {
58                            p { class: "text-base mt-2 opacity-70", " OU" }
59                            p { class: "text-lg mt-2", " {alternative}" }
60                        }
61                    }
62                }
63            } else {
64                p { class: "text-purple-300", "Chargement..." }
65            }
66
67            // Simple button to call load_next
68            div { class: "flex gap-6",
69
70                button {
71                    class: "px-6 py-3 rounded-full bg-green-500/80 hover:scale-110 transition shadow-lg",
72                    onclick: load_next,
73                    "Passer"
74                }
75            }
76        }
77    }
78}
