1use crate::components::navigation::ArrowButton;
2use crate::config::state::PLAYERS;
3use dioxus::prelude::*;
4use dioxus_free_icons::icons::fa_solid_icons::FaPlus;
5use dioxus_free_icons::Icon;

7/// Screen used to choose the players
8/// Uses a copy of the player list before applying it while saving
9#[component]
10pub fn PlayerChoice() -> Element {
11    let nav = use_navigator();
12
13    // local player list
14    let mut players = use_signal(|| PLAYERS.read().clone());
15
16    let mut input = use_signal(|| String::new());
17    let mut editing_index = use_signal(|| None::<usize>);
18    let mut show_toast = use_signal(|| false);
19
20    rsx! {
21        section { class: "min-h-screen text-white p-6",
22
23            ArrowButton { onclick: move |_| nav.go_back() }
24
25            section { class: "w-full max-w-md mx-auto flex flex-col items-center gap-6 mt-6",
26
27                h1 { class: "text-3xl font-bold text-purple-300", "Joueurs" }
28
29                // Player input
30                div { class: "flex w-full gap-2",
31
32                    input {
33                        class: "flex-1 px-4 py-2 rounded-full bg-purple-800/40 border border-purple-500/40 focus:outline-none focus:ring-2 focus:ring-purple-400 placeholder:text-purple-300",
34                        placeholder: "Ajouter un joueur...",
35                        value: "{input}",
36                        oninput: move |e| input.set(e.value()),
37                    }
38
39                    button {
40                        class: "px-4 py-2 rounded-full bg-purple-600 hover:bg-purple-500 active:scale-95 transition-all shadow-[0_0_10px_rgba(168,85,247,0.8)]",
41                        onclick: move |_| {
42                            let name = input().trim().to_string();
43
44                            if !name.is_empty() {
45                                players.write().push(name);
46                                input.set(String::new());
47                            }
48                        },
49                        Icon {
50                            width: 30,
51                            height: 30,
52                            fill: "black",
53                            icon: FaPlus,
54                        }
55                    }
56                }
57
58                // Player list
59                div { class: "w-full flex flex-col gap-3",
60
61                    for (index , name) in players().iter().enumerate() {
62                        div { class: "flex items-center justify-between px-4 py-3 rounded-2xl bg-purple-800/30 border border-purple-500/30 shadow-[0_0_12px_rgba(168,85,247,0.3)] backdrop-blur",
63
64                            if editing_index() == Some(index) {
65                                input {
66                                    class: "flex-1 bg-transparent outline-none text-purple-200",
67                                    value: "{name}",
68                                    oninput: move |e| {
69                                        players.write()[index] = e.value();
70                                    },
71                                }
72
73                                button {
74                                    class: "ml-2 text-green-400 hover:text-green-300",
75                                    onclick: move |_| editing_index.set(None),
76                                    "✔"
77                                }
78                            } else {
79                                p { class: "flex-1 text-lg text-purple-200", "{name}" }
80
81                                button {
82                                    class: "ml-2 text-yellow-400 hover:text-yellow-300",
83                                    onclick: move |_| editing_index.set(Some(index)),
84                                    "✏️"
85                                }
86                            }
87
88                            button {
89                                class: "ml-2 text-red-400 hover:text-red-300",
90                                onclick: move |_| {
91                                    players.write().remove(index);
92                                },
93                                "❌"
94                            }
95                        }
96                    }
97                }
98
99                // SAVE
100                button {
101                    class: "mt-4 w-full py-3 rounded-full bg-gradient-to-r from-purple-600 to-pink-500 hover:scale-105 active:scale-95 transition-all font-bold",
102
103                    onclick: move |_| {
104                        let list = players();
105
106                        *PLAYERS.write() = list;
107
108                        show_toast.set(true);
109
110                        spawn(async move {
111                            use tokio::time::{sleep, Duration};
112                            sleep(Duration::from_secs(2)).await;
113                            show_toast.set(false);
114                        });
115                    },
116
117                    "Sauvegarder"
118                }
119
120                // TOAST
121                if show_toast() {
122                    div { class: "fixed bottom-6 left-1/2 -translate-x-1/2 px-6 py-3 rounded-full bg-purple-600 text-white animate-[fadeIn_0.3s_ease]",
123                        "Joueurs enregistrés !"
124                    }
125                }
126            }
127        }
128    }
129}
